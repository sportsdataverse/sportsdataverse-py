# 0.0.54 — Per-Sport Return Schemas + `return_parsed=True` Default Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `0.0.54` with (A) per-sport-correct `@return` schemas derived by running the real parsers against per-sport captured fixtures, and (B) `return_parsed=True` as the default for parser-backed wrappers.

**Architecture:** Today six ESPN endpoint types share one **sport-agnostic** `returns_schema` (`scoreboard`/`summary`/`teams`/`team_roster`/`standings`/`leaders`), each `scope: universal`, so one schema authored from a single sport drives the `@return` table for all 8 leagues — wrong for 7 of 8 wherever payloads differ by sport (worst: `summary` boxscore frames). Native APIs (`nhl_*`, `mlb_api`) carry no return schema at all. Fix: a new `generate.py --schemas` step runs each parser on the per-league captured fixture and emits **per-league** column schemas (`schemas/{name}/{league}.yaml`), mirroring the proven `--loader-schemas` parquet-footer introspection. The generic `schemas/{name}.yaml` is repurposed as the hand-curated **description dictionary** (descriptions merged in by column name) and the fallback for leagues without a fixture. Then `return_parsed` flips to `True` in the two wrapper templates. The drift gate (`generate.py --check`) keeps everything consistent.

**Tech Stack:** Python 3.9–3.14, Jinja2 codegen (`tools/codegen/`), polars parsers (`sportsdataverse/_common_espn_parsers.py` + native `*_parsers.py`), pytest, Docusaurus, `uv`.

**Verified facts (2026-06-08):**
- `_return_table(schema_name)` at `tools/codegen/generate.py:127-149` resolves `schemas/{schema_name}.yaml`; handles `kind: dataframe` (top-level `columns`) and `kind: frames` (`frames:[{section,columns}]`). It is called at `generate.py:219` as `self.return_table = _return_table(ep.returns_schema)` in `_EndpointView.__init__` (which receives `league`).
- ESPN endpoints declaring `returns_schema`: `espn_site_v2.yaml` (scoreboard:26, summary:37, teams:96, team_roster:115, standings:252), `espn_core_v2.yaml` (teams:888, standings:1020), `espn_web_v3.yaml` (leaders:82). All `scope: universal` → all 8 leagues (`leagues.yaml:7-14`: nba, wnba, mbb, wbb, cfb, nfl, mlb, nhl).
- Parser signatures (all in `_common_espn_parsers.py`): `parse_scoreboard/parse_teams/parse_standings/parse_leaders/parse_team_roster(payload, return_as_pandas=False) -> pl.DataFrame`; `parse_summary(payload, section=None, return_as_pandas=False)` → **dict of 17 frames** keyed by `SUMMARY_SECTION_PARSERS` when `section=None`.
- Per-league fixtures present: `tests/fixtures/espn/summary_{league}.json` (8), `team_roster_{league}.json` (8). **Missing** (need capture): `scoreboard/teams/standings/leaders × 8 = 32`.
- Native fixtures present: `tests/fixtures/{mlb_api,nhl_api_web,nhl_edge,nhl_records,nhl_stats_rest}/*.json` (8/17/7/6/8) — named by their own convention (e.g. `roster_edm_2024.json`), so they need a fixture→endpoint map (from each dir's `README.md` provenance).

- Capture-script template: `tools/capture_cfb_fixtures.py`.

```python
"""One-off: capture per-league ESPN payloads for the endpoints whose return
schema is sport-specific (scoreboard/teams/standings/leaders) so the schema
introspector (generate.py --schemas) can derive per-sport columns offline.

Run once WITH network:  python -m tools.capture_espn_fixtures
Commit the resulting tests/fixtures/espn/{endpoint}_{league}.json files.
"""
from __future__ import annotations

import importlib
import json
from pathlib import Path

OUT = Path("tests/fixtures/espn")
LEAGUES = ["nba", "wnba", "mbb", "wbb", "cfb", "nfl", "mlb", "nhl"]
# (endpoint stem, wrapper short, example kwargs) — wrappers called with
# return_parsed=False to capture the RAW payload the parser consumes.
ENDPOINTS = [
    ("scoreboard", "scoreboard", {}),
    ("teams", "teams", {}),
    ("standings", "standings", {"season": 2024}),
    ("leaders", "leaders", {"season": 2024}),
]


def main() -> None:
    OUT.mkdir(parents=True, exist_ok=True)
    for league in LEAGUES:
        mod = importlib.import_module(f"sportsdataverse.{league}")
        for stem, short, kwargs in ENDPOINTS:
            fn = getattr(mod, f"espn_{league}_{short}", None)
            if fn is None:
                print(f"skip {league}/{short}: no wrapper")
                continue
            try:
                raw = fn(return_parsed=False, **kwargs)
                path = OUT / f"{stem}_{league}.json"
                path.write_text(json.dumps(raw), encoding="utf-8")
                print(f"wrote {path} ({path.stat().st_size} bytes)")
            except Exception as e:  # noqa: BLE001
                print(f"FAIL {league}/{short}: {e}")


if __name__ == "__main__":
    main()
```

```python
# tools/codegen/generate.py  (place near refresh_loader_schemas)

# returns_schema name -> (parser callable, "frames" | "dataframe").
# parse_summary returns a {section: frame} dict; the rest return one frame.
def _return_schema_parsers():
    from sportsdataverse import _common_espn_parsers as P
    return {
        "scoreboard": (P.parse_scoreboard, "dataframe"),
        "teams": (P.parse_teams, "dataframe"),
        "team_roster": (P.parse_team_roster, "dataframe"),
        "standings": (P.parse_standings, "dataframe"),
        "leaders": (P.parse_leaders, "dataframe"),
        "summary": (P.parse_summary, "frames"),
    }


def _desc_lookup(schema_name: str) -> dict:
    """{column_name: description} from the hand-curated generic schema, used to
    annotate the introspected per-league columns (which carry only name+type)."""
    import yaml

    p = ROOT / "tools" / "codegen" / "schemas" / f"{schema_name}.yaml"
    if not p.exists():
        return {}
    d = yaml.safe_load(p.read_text(encoding="utf-8")) or {}
    out = {}
    if d.get("kind") == "frames":
        for blk in d.get("frames", []):
            for c in blk.get("columns", []):
                out[c["name"]] = c.get("description", "")
    else:
        for c in d.get("columns", []):
            out[c["name"]] = c.get("description", "")
    return out


def _cols_from_frame(df, descs: dict) -> list:
    return [
        {"name": n, "type": _pl_to_doc_type(str(t)), "description": descs.get(n, "")}
        for n, t in zip(df.columns, df.dtypes)
    ]


def _pl_to_doc_type(pl_type: str) -> str:
    s = pl_type.lower()
    if "int" in s:
        return "integer"
    if "float" in s:
        return "double"
    if "bool" in s:
        return "logical"
    return "character"


_LEAGUES = ["nba", "wnba", "mbb", "wbb", "cfb", "nfl", "mlb", "nhl"]
_FIX = ROOT / "tests" / "fixtures" / "espn"


def refresh_return_schemas() -> int:
    """Run each parser on every per-league captured fixture and emit per-league
    column schemas under schemas/{name}/{league}.yaml. Descriptions are merged
    from the hand-curated generic schemas/{name}.yaml by column name."""
    import json

    import yaml

    written = skipped = 0
    for name, (parser, kind) in _return_schema_parsers().items():
        descs = _desc_lookup(name)
        outdir = ROOT / "tools" / "codegen" / "schemas" / name
        for league in _LEAGUES:
            fx = _FIX / f"{name}_{league}.json"
            if not fx.exists():
                skipped += 1
                continue
            payload = json.loads(fx.read_text(encoding="utf-8"))
            outdir.mkdir(parents=True, exist_ok=True)
            if kind == "frames":
                frames = parser(payload)  # {section: df}
                doc = {
                    "schema": name,
                    "kind": "frames",
                    "frames": [
                        {"section": sec, "columns": _cols_from_frame(df, descs)}
                        for sec, df in frames.items()
                    ],
                }
            else:
                df = parser(payload)
                doc = {"schema": name, "kind": "dataframe", "columns": _cols_from_frame(df, descs)}
            (outdir / f"{league}.yaml").write_text(yaml.safe_dump(doc, sort_keys=False, width=120), encoding="utf-8")
            written += 1
    print(f"return schemas: {written} per-league written, {skipped} skipped (no fixture)")
    return 0
```
